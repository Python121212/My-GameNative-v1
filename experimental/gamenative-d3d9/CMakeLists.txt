cmake_minimum_required(VERSION 3.16)
project(gamenative_d3d9_experimental LANGUAGES CXX)
find_package(Vulkan REQUIRED)
add_library(gn_upload STATIC src/upload.cpp src/texture9.cpp)
target_compile_features(gn_upload PUBLIC cxx_std_17)
target_include_directories(gn_upload PUBLIC src)
target_link_libraries(gn_upload PUBLIC Vulkan::Vulkan)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(gn_upload PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(upload_test tests/upload_test.cpp)
  target_link_libraries(upload_test PRIVATE gn_upload)
  add_test(NAME upload COMMAND upload_test)
  add_executable(texture9_test tests/texture9_test.cpp)
  target_link_libraries(texture9_test PRIVATE gn_upload)
  add_test(NAME texture9 COMMAND texture9_test)
endif()
